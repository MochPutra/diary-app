import React from 'react';
import { View, Text, Image, StyleSheet, Pressable } from 'react-native';
import { COLORS, SPACE, RADIUS, TYPO, MOODS } from '../styles/tokens';
import { toSource } from '../utils/imageSource';

export default function DiaryCard({
  title, date, preview, moodUri, mood = 'calm', onPress, variant = 'list',
}) {
  const moodInfo = MOODS[mood] || MOODS.calm;

  return (
    <Pressable
      onPress={onPress}
      android_ripple={{ color: COLORS.border }}
      style={({ pressed }) => [{ opacity: pressed ? 0.85 : 1 }]}
      accessible
      accessibilityRole="button"
      accessibilityLabel={`Entri ${title} tanggal ${date}, mood ${moodInfo.label}`}
      accessibilityHint="Ketuk untuk membuka detail entri"
    >
      <View
        style={[
          styles.card,
          variant === 'grid' && styles.cardGrid,
          { borderColor: moodInfo.color },
        ]}
      >
        {moodUri ? (
          <Image source={toSource(moodUri)} style={styles.mood} />
        ) : (
          <View style={[styles.mood, styles.moodFallback, { backgroundColor: moodInfo.color }]}>
            <Text style={styles.moodInitial}>{title.charAt(0).toUpperCase()}</Text>
          </View>

        <View style={styles.content}>
          <Text style={styles.title} numberOfLines={1}>{title}</Text>
          <Text style={styles.date}>{date}</Text>
          <Text style={styles.preview} numberOfLines={3} ellipsizeMode="tail">
            {preview}
          </Text>
          <View style={[styles.badge, { borderColor: moodInfo.color }]}>
            <Text style={[styles.badgeText, { color: moodInfo.color }]}>
              {moodInfo.label}
            </Text>
          </View>
        </View>
      </View>
    </Pressable>
  );

import React, { useEffect, useRef, useState } from 'react';
import {
  Modal, View, Text, TextInput, Pressable, ScrollView, Animated,
  KeyboardAvoidingView, Platform, StyleSheet,
} from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import MoodChip from './MoodChip';
import { COLORS, SPACE, RADIUS, TYPO, MOODS, SIZE } from '../styles/tokens';

const TITLE_MAX = 50;
const BODY_MAX = 500;
const BODY_MIN = 10;

export default function EntryForm({ visible, onClose, onSubmit }) {
  const [title, setTitle] = useState('');
  const [body, setBody] = useState('');
  const [mood, setMood] = useState('calm');
  const [focused, setFocused] = useState(null);
  const [submitted, setSubmitted] = useState(false);
  const shake = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    if (visible) {
      setTitle('');
      setBody('');
      setMood('calm');
      setSubmitted(false);
    }
  }, [visible]);

  const titleError = title.trim().length === 0 ? 'Judul wajib diisi.' : null;
  const bodyError =
    body.trim().length < BODY_MIN ? `Cerita minimal ${BODY_MIN} karakter.` : null;
  const moodInfo = MOODS[mood];

  const runShake = () => {
    Animated.sequence(
      [10, -10, 6, -6, 0].map((v) =>
        Animated.timing(shake, { toValue: v, duration: 50, useNativeDriver: true })
      )
    ).start();
  };

  const handleSave = () => {
    setSubmitted(true);
    if (titleError || bodyError) {
      runShake();
      return;
    }
    onSubmit({ title: title.trim(), body: body.trim(), mood });
    onClose();
  };

  const inputStyle = (field, error) => [
    styles.input,
    focused === field && styles.inputFocused,
    submitted && error && styles.inputError,
  ];

  return (
    <Modal visible={visible} animationType="slide" onRequestClose={onClose}>
      <SafeAreaView style={styles.container} edges={['top', 'bottom']}>
        <View style={[styles.moodBar, { backgroundColor: moodInfo.color }]} />

        <View style={styles.header}>
          <Text style={styles.heading} accessibilityRole="header">Entri baru</Text>
          <Pressable
            onPress={onClose}
            style={styles.cancel}
            accessibilityRole="button"
            accessibilityLabel="Batal, tutup formulir"
          >
            <Text style={styles.cancelText}>Batal</Text>
          </Pressable>
        </View>

        <KeyboardAvoidingView
          style={styles.flex}
          behavior={Platform.OS === 'ios' ? 'padding' : undefined}
        >
          <ScrollView contentContainerStyle={styles.form} keyboardShouldPersistTaps="handled">
            <View style={styles.field}>
              <Text style={styles.label}>Judul</Text>
              <TextInput
                value={title}
                onChangeText={setTitle}
                onFocus={() => setFocused('title')}
                onBlur={() => setFocused(null)}
                maxLength={TITLE_MAX}
                placeholder="Contoh: Pagi yang tenang"
                placeholderTextColor={COLORS.muted}
                style={inputStyle('title', titleError)}
                accessibilityLabel="Judul entri"
              />
              <View style={styles.meta}>
                <Text style={styles.error} accessibilityLiveRegion="polite">
                  {submitted && titleError ? titleError : ' '}
                </Text>
                <Text style={styles.counter}>{title.length}/{TITLE_MAX}</Text>
              </View>
            </View>

            <View style={styles.field}>
              <Text style={styles.label}>Cerita hari ini</Text>
              <TextInput
                value={body}
                onChangeText={setBody}
                onFocus={() => setFocused('body')}
                onBlur={() => setFocused(null)}
                maxLength={BODY_MAX}
                multiline
                textAlignVertical="top"
                placeholder="Apa yang kamu alami hari ini?"
                placeholderTextColor={COLORS.muted}
                style={[...inputStyle('body', bodyError), styles.textarea]}
                accessibilityLabel="Isi cerita entri"
              />
              <View style={styles.meta}>
                <Text style={styles.error} accessibilityLiveRegion="polite">
                  {submitted && bodyError ? bodyError : ' '}
                </Text>
                <Text style={[styles.counter, body.length >= BODY_MAX * 0.9 && styles.counterWarn]}>
                  {body.length}/{BODY_MAX}
                </Text>
              </View>
            </View>

            <View style={styles.field}>
              <Text style={styles.label}>Suasana hati</Text>
              <View style={styles.moods}>
                {Object.entries(MOODS).map(([key, m]) => (
                  <MoodChip
                    key={key}
                    label={m.label}
                    a11yLabel={`Pilih mood ${m.label}`}
                    color={m.color}
                    active={mood === key}
                    onPress={() => setMood(key)}
                  />
                ))}
              </View>
            </View>

            <Animated.View style={{ transform: [{ translateX: shake }] }}>
              <Pressable
                onPress={handleSave}
                style={({ pressed }) => [
                  styles.save,
                  { backgroundColor: moodInfo.color },
                  pressed && styles.savePressed,
                ]}
                accessibilityRole="button"
                accessibilityLabel="Simpan entri"
              >
                <Text style={styles.saveText}>Simpan entri</Text>
              </Pressable>
            </Animated.View>
          </ScrollView>
        </KeyboardAvoidingView>
      </SafeAreaView>
    </Modal>
  );
}

const styles = StyleSheet.create({

  container: { flex: 1, backgroundColor: COLORS.surface },
  moodBar: { height: 4 },
  header: {
    flexDirection: 'row',
    alignItems: 'center',

  },
  heading: { fontSize: TYPO.h1, fontWeight: '700', color: COLORS.text },

import React, { useRef } from 'react';
import { View, Text, TextInput, Pressable, ScrollView, Animated, StyleSheet } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import MoodChip from './MoodChip';
import { COLORS, SPACE, RADIUS, TYPO, MOODS, SIZE } from '../styles/tokens';

export default function SearchBar({ value, onChangeText, mood, onMoodChange, resultCount }) {
  const focus = useRef(new Animated.Value(0)).current;

  const animateTo = (to) =>
    Animated.timing(focus, { toValue: to, duration: 180, useNativeDriver: false }).start();

  const borderColor = focus.interpolate({
    inputRange: [0, 1],
    outputRange: [COLORS.border, COLORS.accent],
  });

  return (
    <View style={styles.wrap}>
      <Animated.View style={[styles.box, { borderColor }]}>
        <Ionicons name="search" size={20} color={COLORS.muted} />
        <TextInput
          value={value}
          onChangeText={onChangeText}
          onFocus={() => animateTo(1)}
          onBlur={() => animateTo(0)}
          placeholder="Cari judul atau isi entri"
          placeholderTextColor={COLORS.muted}
          style={styles.input}
          returnKeyType="search"
          autoCorrect={false}
          accessibilityLabel="Cari entri diary"
        />
        {value.length > 0 && (
          <Pressable
            onPress={() => onChangeText('')}
            hitSlop={8}
            style={styles.clear}
            accessibilityRole="button"
            accessibilityLabel="Hapus pencarian"
          >
            <Ionicons name="close-circle" size={22} color={COLORS.muted} />
          </Pressable>
        )}
      </Animated.View>

      <ScrollView
        horizontal
        showsHorizontalScrollIndicator={false}
        keyboardShouldPersistTaps="handled"
        contentContainerStyle={styles.chips}
      >
        <MoodChip
          label="Semua"
          a11yLabel="Tampilkan semua mood"
          active={mood === null}
          onPress={() => onMoodChange(null)}
        />
        {Object.entries(MOODS).map(([key, m]) => (
          <MoodChip
            key={key}
            label={m.label}
            a11yLabel={`Filter mood ${m.label}`}
            color={m.color}
            active={mood === key}
            onPress={() => onMoodChange(mood === key ? null : key)}
          />
        ))}
      </ScrollView>

      <Text style={styles.count} accessibilityLiveRegion="polite">
        {resultCount} entri ditemukan
      </Text>
    </View>
  );
}

const styles = StyleSheet.create({
  wrap: { paddingHorizontal: SPACE.l, paddingTop: SPACE.m, gap: SPACE.s },
  box: {
    flexDirection: 'row',
    alignItems: 'center',
    gap: SPACE.s,
    minHeight: SIZE.touch,
    paddingHorizontal: SPACE.m,
    borderWidth: 1.5,
    borderRadius: RADIUS.card,
    backgroundColor: COLORS.surface,
  },
  input: { flex: 1, fontSize: TYPO.title, color: COLORS.text, paddingVertical: SPACE.s },
  clear: { alignItems: 'center', justifyContent: 'center' },
  chips: { gap: SPACE.s, paddingVertical: SPACE.xs },
  count: { fontSize: TYPO.caption, color: COLORS.muted },
});

              {hasFilter
                ? 'Coba kata kunci lain atau hapus filter mood.'
                : 'Ketuk "Tulis entri" untuk membuat catatan pertamamu.'}
            </Text>
            {hasFilter && (
              <Pressable
                onPress={resetFilter}
                style={styles.resetBtn}
                accessibilityRole="button"
                accessibilityLabel="Hapus pencarian dan filter"
              >
                <Text style={styles.resetText}>Hapus filter</Text>
              </Pressable>
            )}
          </View>
        }
      />

      <Pressable
        onPress={() => setFormVisible(true)}
        style={({ pressed }) => [styles.fab, pressed && styles.fabPressed]}
        accessibilityRole="button"
        accessibilityLabel="Tulis entri baru"
      >
        <Ionicons name="add" size={24} color={COLORS.onAccent} />
        <Text style={styles.fabText}>Tulis entri</Text>
      </Pressable>

      <DiaryDetailModal
        visible={detailVisible}
        entry={selected}
        onClose={() => setDetailVisible(false)}
      />
      <EntryForm
        visible={formVisible}
        onClose={() => setFormVisible(false)}
        onSubmit={addEntry}
      />
    </View>
  );
}
